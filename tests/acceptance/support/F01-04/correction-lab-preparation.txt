from pathlib import Path
import tempfile,shutil,hashlib,json
source=Path('/Users/javiercamaraportepetit/vexa/.runtime/closure-1789847917909284000/integrated')
lab=Path(tempfile.mkdtemp(prefix='vexa-f01-04-authlab-'))
exclude={'.git','.next','node_modules','private','.runtime'}
def ignore(p,names):return [n for n in names if n in exclude or n.startswith('.env')]
for rel in ['apps','packages','package.json','package-lock.json']:
 p=source/rel
 if p.is_dir():shutil.copytree(p,lab/rel,ignore=ignore)
 else:shutil.copy2(p,lab/rel)
(lab/'supabase/migrations').mkdir(parents=True)
for p in Path('/Users/javiercamaraportepetit/vexa/supabase/migrations').glob('000[1-4]*.sql'):shutil.copy2(p,lab/'supabase/migrations'/p.name)
Path('/tmp/f01-lab-path').write_text(str(lab))
def hashes(root):
 result={}
 for p in root.rglob('*'):
  if any(x in exclude or x.startswith('.env') for x in p.relative_to(root).parts):continue
  if p.is_file():result[str(p.relative_to(root))]=hashlib.sha256(p.read_bytes()).hexdigest()
 return result
manifest={'source':str(source),'lab':str(lab),'source_hashes':hashes(source),'lab_hashes':hashes(lab)}
Path('tests/acceptance/support/F01-04/correction-immutability.json').write_text(json.dumps(manifest,indent=2)+'\n')
print(lab)
